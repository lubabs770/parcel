**Goal:** Build `parcel`, a light, fast, good-looking Rust TUI mail client that drives any JSON-speaking email CLI (himalaya by default) and follows the Omarchy theme live.

**Architecture:** A `backend` layer turns per-action command templates from config into argv, runs them with no shell, and extracts envelopes from JSON via tiny dot-paths. Config is a directory of small optional TOML files layered over compiled-in presets. The ratatui TUI is one event loop fed by a single `mpsc` channel (input thread, ticker, SIGUSR1, and one short-lived thread per backend job); triage is optimistic with revert on failure.

**Tech Stack:** Rust 2024 (rustc 1.95), ratatui 0.30 (crossterm via `ratatui::crossterm`), clap 4.6 derive, serde 1, serde_json 1, toml 1.1, signal-hook 0.4 (unix), dev: tempfile 3, insta 1.49.

```rust
//! Command templates → argv. No shell is ever involved.

use std::collections::HashMap;

/// Expand a command template into argv. The template is split on whitespace
/// first; each `{name}` is then replaced inside its own argv entry, so a value
/// with spaces or shell metacharacters stays one argument. `{name...}` splits
/// the value on whitespace into several arguments. A word that is exactly one
/// placeholder with an empty value is dropped.
pub fn expand(template: &str, vars: &HashMap<&str, String>) -> Result<Vec<String>, String> {
    let mut argv = Vec::new();
    for word in template.split_whitespace() {
        if let Some(name) = word.strip_prefix('{').and_then(|w| w.strip_suffix("...}")) {
            let v = vars.get(name).ok_or_else(|| format!("unknown placeholder {{{name}...}}"))?;
            argv.extend(v.split_whitespace().map(String::from));
            continue;
        }
        let mut out = String::new();
        let mut rest = word;
        while let Some(start) = rest.find('{') {
            let Some(len) = rest[start..].find('}') else { break };
            let name = &rest[start + 1..start + len];
            let v = vars.get(name).ok_or_else(|| format!("unknown placeholder {{{name}}}"))?;
            out.push_str(&rest[..start]);
            out.push_str(v);
            rest = &rest[start + len + 1..];
        }
        out.push_str(rest);
        let whole = word.starts_with('{') && word.ends_with('}');
        if !(out.is_empty() && whole) {
            argv.push(out);
        }
    }
    Ok(argv)
}
```

```rust
//! Tiny JSON paths: "from.0.name", with "a|b" fallbacks.

use serde_json::Value;

/// Look up a dot path in a JSON value; numeric segments index arrays. `a|b`
/// tries each alternative and returns the first non-null, non-empty hit.
/// The empty path returns `v` itself (for CLIs that print a bare array).
pub fn get<'a>(v: &'a Value, path: &str) -> Option<&'a Value> {
    path.split('|').find_map(|alt| {
        let hit = alt.trim().split('.').filter(|s| !s.is_empty()).try_fold(v, |cur, key| match cur {
            Value::Object(m) => m.get(key),
            Value::Array(a) => key.parse::<usize>().ok().and_then(|i| a.get(i)),
            _ => None,
        })?;
        match hit {
            Value::Null => None,
            Value::String(s) if s.is_empty() => None,
            _ => Some(hit),
        }
    })
}

/// `get` as text: strings as-is, anything else via its JSON rendering.
pub fn text(v: &Value, path: &str) -> Option<String> {
    get(v, path).map(|h| match h {
        Value::String(s) => s.clone(),
        other => other.to_string(),
    })
}
```

```rust
//! Config is a directory of small optional files layered over compiled-in

pub struct Config {

impl Config {
    pub fn load() -> Result<Self, String> {
        Self::load_from(&config_dir())
    }

    pub fn load_from(dir: &Path) -> Result<Self, String> {
        let settings: Settings = layered(DEFAULT_CONFIG, &dir.join("config.toml"))?;
        let path = dir.join("backends").join(format!("{}.toml", settings.backend));
        let preset = preset(&settings.backend);
        if preset.is_none() && !path.exists() {
            return Err(format!("backend {:?}: no built-in preset and no {}", settings.backend, path.display()));
        }
        let backend = layered(preset.unwrap_or(""), &path)?;
        let keys = read_table(&dir.join("keys.toml"))?;
        Ok(Self { dir: dir.to_path_buf(), settings, backend, keys })
    }
}

pub fn preset(name: &str) -> Option<&'static str> {
    (name == "himalaya").then_some(HIMALAYA)
}

/// Parse `base`, overlay the file at `path` (if any) key by key, deserialize.
fn layered<T: DeserializeOwned>(base: &str, path: &Path) -> Result<T, String> {
    let mut table: Table = toml::from_str(base).map_err(|e| format!("built-in default: {e}"))?;
    merge(&mut table, read_table(path)?);
    // If toml 1.x lacks Value::try_into, use `T::deserialize(Value::Table(table))`.
    Value::Table(table).try_into().map_err(|e| format!("{}: {e}", path.display()))
}

/// A TOML file as a table; a missing file is an empty table.
pub fn read_table(path: &Path) -> Result<Table, String> {
    match fs::read_to_string(path) {
        Ok(s) => toml::from_str(&s).map_err(|e| format!("{}: {e}", path.display())),
        Err(e) if e.kind() == io::ErrorKind::NotFound => Ok(Table::new()),
        Err(e) => Err(format!("{}: {e}", path.display())),
    }
}

/// Deep-merge `over` into `base`: nested tables merge, anything else replaces.
pub fn merge(base: &mut Table, over: Table) {
    for (k, v) in over {
        match (base.get_mut(&k), v) {
            (Some(Value::Table(b)), Value::Table(o)) => merge(b, o),
            (_, v) => {
                base.insert(k, v);
            }
        }
    }
}

/// Write the defaults out for editing. Never overwrites an existing file.
pub fn init(dir: &Path) -> Result<Vec<PathBuf>, String> {
    let files = [("config.toml", DEFAULT_CONFIG), ("keys.toml", DEFAULT_KEYS), ("backends/himalaya.toml", HIMALAYA)];
    let mut written = Vec::new();
    for (rel, body) in files {
        let path = dir.join(rel);
        if path.exists() {
            continue;
        }
```

```rust
//! Run a backend command (argv, never a shell) and log failures.

use std::{fs::{self, OpenOptions}, io::Write, path::PathBuf, process::{Command, Stdio}, time::{SystemTime, UNIX_EPOCH}};

/// Ok(stdout) on success. On failure: Err(first non-empty stderr line), with
/// the full stderr appended to parcel.log.
pub fn run(argv: &[String], stdin: Option<&[u8]>) -> Result<String, String> {
    let (bin, args) = argv.split_first().ok_or("empty command")?;
    let mut child = Command::new(bin)
        .args(args)
        .stdin(if stdin.is_some() { Stdio::piped() } else { Stdio::null() })
        .stdout(Stdio::piped())
        .stderr(Stdio::piped())
        .spawn()
        .map_err(|e| format!("{bin}: {e}"))?;
    if let (Some(input), Some(mut pipe)) = (stdin, child.stdin.take()) {
        pipe.write_all(input).map_err(|e| format!("{bin}: {e}"))?;
    } // pipe dropped here → EOF for the child
    let out = child.wait_with_output().map_err(|e| format!("{bin}: {e}"))?;
    if out.status.success() {
        return Ok(String::from_utf8_lossy(&out.stdout).into_owned());
    }
    let err = String::from_utf8_lossy(&out.stderr);
    log(argv, &err);
    Err(err
        .lines()
        .map(str::trim)
        .find(|l| !l.is_empty())
        .map(String::from)
        .unwrap_or_else(|| format!("{bin} exited with {}", out.status)))
}

pub fn log_path() -> PathBuf {
    crate::config::state_dir().join("parcel.log")
}

fn log(argv: &[String], stderr: &str) {
    let path = log_path();
    let _ = fs::create_dir_all(path.parent().unwrap());
    if let Ok(mut f) = OpenOptions::new().create(true).append(true).open(path) {
        let ts = SystemTime::now().duration_since(UNIX_EPOCH).map(|d| d.as_secs()).unwrap_or(0);
        let _ = writeln!(f, "[{ts}] {}\n{}", argv.join(" "), stderr.trim_end());
    }
}
```

```rust
        exec::run(&argv, stdin.map(str::as_bytes)).map_err(|e| format!("{action}: {e}"))
    }

        let out = self.call(action, vars, None)?;
        serde_json::from_str(&out).map_err(|e| format!("{action}: output is not JSON ({e})"))
    }

        fields::get(v, path).and_then(Value::as_array).ok_or_else(|| format!("{action}: no array at `{path}`"))
    }

        let f = &self.spec.fields;
        Self::items(action, v, &f.items)?
            .iter()
            .map(|item| {
                let id = fields::text(item, &f.id).ok_or_else(|| format!("{action}: item has no `{}`", f.id))?;

                        a.iter()

                                Value::String(s) => Some(s.clone()),
                                other => fields::text(other, &f.flag_name),
                            })
                            .collect()

                    .unwrap_or_default();
                let has = |name: &str| !name.is_empty() && flags.iter().any(|x| x.eq_ignore_ascii_case(name));
                Ok(Envelope {
                    id,
                    subject: fields::text(item, &f.subject).unwrap_or_default(),

                    flagged: has(&self.spec.flagged_flag),
                })
            })
            .collect()

        self.call("read", &[("mailbox", mailbox), ("id", id)], None)

            Flag::Seen => &self.spec.seen_flag,
            Flag::Flagged => &self.spec.flagged_flag,
        };

    pub fn send(&self, raw: &str) -> Result<(), String> {
        self.call("send", &[], Some(raw)).map(drop)

            return Ok("To: \nCc: \nSubject: \n\n".into());
        }
        self.call(action, &[("mailbox", mailbox), ("id", id)], None)
    }
```

```rust
use crate::config::Config;
use std::{env, path::Path};

pub fn run() -> Result<(), String> {
    let mut ok = true;
    let mut check = |pass: bool, what: String| {
        println!("{} {what}", if pass { "✓" } else { "✗" });
        ok &= pass;
    };
    let cfg = match Config::load() {
        Ok(c) => {
            check(true, format!("config     {}", c.dir.display()));
            c
        }
        Err(e) => {
            check(false, format!("config     {e}"));
            return Err("doctor found problems".into());
        }
    };
    let be = Backend::new(cfg.backend.clone(), cfg.settings.page_size);
    for action in Backend::ACTIONS {
        match be.template(action) {
            Some(t) => {
                let bin = t.split_whitespace().next().unwrap_or("");
                check(on_path(bin), format!("{action:<10} {bin}"));

}
```

```rust
pub struct Theme {
    pub bg: Color,

    pub fg: Color,

/// themes/<name>.toml: the 12 tokens as hex strings.
#[derive(Deserialize)]
struct ThemeFile {
    bg: String,
    surface: String,
    overlay: String,
    fg: String,
    muted: String,
    accent: String,
    unread: String,
    flagged: String,
    danger: String,
    ok: String,
    selection: String,
    border: String,
}

impl Theme {
    /// Tokens in field order: bg surface overlay fg muted accent unread flagged danger ok selection border.
    pub fn from_hex(h: [&str; 12]) -> Result<Self, String> {
        let c = |i: usize| rgb(h[i]).map(|(r, g, b)| Color::Rgb(r, g, b)).ok_or_else(|| format!("bad colour {:?}", h[i]));
        Ok(Self {
            bg: c(0)?, surface: c(1)?, overlay: c(2)?, fg: c(3)?, muted: c(4)?, accent: c(5)?,
            unread: c(6)?, flagged: c(7)?, danger: c(8)?, ok: c(9)?, selection: c(10)?, border: c(11)?,
        })
    }
}

/// "#rrggbb" or "0xrrggbb".
pub fn rgb(s: &str) -> Option<(u8, u8, u8)> {
    let h = s.trim().trim_start_matches('#').trim_start_matches("0x");
    if h.len() != 6 {
        return None;
    }
    let n = u32::from_str_radix(h, 16).ok()?;
    Some(((n >> 16) as u8, (n >> 8) as u8, n as u8))
}

pub fn load(name: &str, config_dir: &Path) -> Result<Theme, String> {
    if name == "auto" {
        return Ok(builtin::default()); // Task 7 makes this follow Omarchy
    }
    if let Some(t) = builtin::get(name) {
        return Ok(t);
    }
    let path = config_dir.join("themes").join(format!("{name}.toml"));
    let s = fs::read_to_string(&path).map_err(|e| format!("theme {name:?}: {e}"))?;
    let f: ThemeFile = toml::from_str(&s).map_err(|e| format!("{}: {e}", path.display()))?;
    Theme::from_hex([
        &f.bg, &f.surface, &f.overlay, &f.fg, &f.muted, &f.accent, &f.unread, &f.flagged, &f.danger, &f.ok,
        &f.selection, &f.border,
    ])
    .map_err(|e| format!("{}: {e}", path.display()))
}

/// "auto", the built-ins, then your themes/*.toml.
pub fn names(config_dir: &Path) -> Vec<String> {
    let mut v: Vec<String> =
        std::iter::once("auto".to_string()).chain(builtin::THEMES.iter().map(|(n, _)| n.to_string())).collect();
    if let Ok(rd) = fs::read_dir(config_dir.join("themes")) {
        let mut user: Vec<String> = rd
            .flatten()
            .map(|e| e.path())
            .filter(|p| p.extension().is_some_and(|x| x == "toml"))
            .filter_map(|p| p.file_stem()?.to_str().map(String::from))
            .filter(|n| !v.contains(n))
            .collect();
        user.sort();
        v.extend(user);
    }
    v
}
```

```rust
/// Map Omarchy's palette onto parcel's tokens; derive any layer it leaves out from the background.
pub fn map(c: &Colors) -> Theme {
    let bg = rgb(&c.background).unwrap_or((0x1a, 0x1b, 0x26));
    let fg = rgb(&c.foreground).unwrap_or((0xc0, 0xca, 0xf5));
    let light = c.mode.eq_ignore_ascii_case("light") || (c.mode.is_empty() && luminance(bg) > 0.5);
    let (deeper, raised) = if light {
        (mix(bg, (0, 0, 0), 0.04), mix(bg, (0, 0, 0), 0.10))
    } else {
        (mix(bg, (0, 0, 0), 0.20), mix(bg, (255, 255, 255), 0.08))
    };
    let pick = |vals: &[&String], def: Rgb| vals.iter().find_map(|v| rgb(v)).unwrap_or(def);
    let accent = pick(&[&c.accent, &c.blue], fg);
    let col = |(r, g, b): Rgb| Color::Rgb(r, g, b);
    Theme {
        bg: col(bg),
        surface: col(pick(&[&c.dark_background], deeper)),
        overlay: col(pick(&[&c.lighter_background], raised)),
        fg: col(fg),
        muted: col(pick(&[&c.muted, &c.dark_foreground], mix(fg, bg, 0.5))),
        accent: col(accent),
        unread: col(pick(&[&c.cyan], accent)),
        flagged: col(pick(&[&c.yellow], accent)),
        danger: col(pick(&[&c.red], accent)),
        ok: col(pick(&[&c.green], accent)),
        selection: col(pick(&[&c.selection], mix(bg, accent, 0.25))),
        border: col(pick(&[&c.lighter_background], raised)),
    }
}

fn mix(a: Rgb, b: Rgb, t: f32) -> Rgb {
    let f = |x: u8, y: u8| (x as f32 + (y as f32 - x as f32) * t).round() as u8;
    (f(a.0, b.0), f(a.1, b.1), f(a.2, b.2))
}

fn luminance((r, g, b): Rgb) -> f32 {
    (0.2126 * r as f32 + 0.7152 * g as f32 + 0.0722 * b as f32) / 255.0
}

/// Append the repaint line to Omarchy's theme-set hook. Ok(false) if already there.
pub fn hook_install(path: &Path) -> Result<bool, String> {
    if hook_installed(path) {
        return Ok(false);
    }
    let mut s = fs::read_to_string(path).unwrap_or_else(|_| "#!/bin/sh\n".into());
    if !s.ends_with('\n') {
        s.push('\n');
    }
    s.push_str(HOOK_LINE);
    s.push('\n');
    fs::create_dir_all(path.parent().unwrap()).map_err(|e| e.to_string())?;
    fs::write(path, s).map_err(|e| format!("{}: {e}", path.display()))?;
    #[cfg(unix)]
    {
        use std::os::unix::fs::PermissionsExt;
        fs::set_permissions(path, fs::Permissions::from_mode(0o755)).map_err(|e| e.to_string())?;
    }
    Ok(true)
}

/// Remove parcel's line. Ok(false) if it wasn't there.
pub fn hook_uninstall(path: &Path) -> Result<bool, String> {
    let Ok(s) = fs::read_to_string(path) else { return Ok(false) };
    if !s.contains("# parcel:") {
        return Ok(false);
    }
    let kept: String = s.lines().filter(|l| !l.contains("# parcel:")).map(|l| format!("{l}\n")).collect();
    fs::write(path, kept).map_err(|e| format!("{}: {e}", path.display()))?;
    Ok(true)
}
```

```rust
//! RFC 3339 → epoch seconds, and friendly relative dates. No date crate needed.

use std::time::{SystemTime, UNIX_EPOCH};

pub fn now() -> i64 {
    SystemTime::now().duration_since(UNIX_EPOCH).map(|d| d.as_secs() as i64).unwrap_or(0)
}

/// "2026-10-25T18:03:56Z", fractional seconds and ±hh:mm offsets allowed.
pub fn parse(s: &str) -> Option<i64> {
    let n = |r: std::ops::Range<usize>| s.get(r)?.parse::<i64>().ok();
    let (y, mo, d) = (n(0..4)?, n(5..7)?, n(8..10)?);
    let (h, mi, se) = (n(11..13)?, n(14..16)?, n(17..19)?);
    let mut rest = s.get(19..)?;
    if let Some(r) = rest.strip_prefix('.') {
        rest = r.trim_start_matches(|c: char| c.is_ascii_digit());
    }
    let off = match rest.as_bytes().first() {
        None | Some(b'Z') | Some(b'z') => 0,
        Some(&sign @ (b'+' | b'-')) => {
            let o = rest.get(1..3)?.parse::<i64>().ok()? * 3600 + rest.get(4..6)?.parse::<i64>().ok()? * 60;
            if sign == b'+' { o } else { -o }
        }
        _ => return None,
    };
    Some(days_from_civil(y, mo, d) * 86400 + h * 3600 + mi * 60 + se - off)
}

/// "now", "5m", "2h", "Tue", "Oct 3", "2024"; unparseable dates are shown as-is.
pub fn relative(s: &str, now: i64) -> String {
    const MON: [&str; 12] = ["Jan", "Feb", "Mar", "Apr", "May", "Jun", "Jul", "Aug", "Sep", "Oct", "Nov", "Dec"];
    const DOW: [&str; 7] = ["Thu", "Fri", "Sat", "Sun", "Mon", "Tue", "Wed"]; // 1970-01-01 was a Thursday
    let Some(t) = parse(s) else { return s.chars().take(10).collect() };
    match now - t {
        a if a < 60 => "now".into(),
        a if a < 3600 => format!("{}m", a / 60),
        a if a < 86400 => format!("{}h", a / 3600),
        a if a < 6 * 86400 => DOW[t.div_euclid(86400).rem_euclid(7) as usize].into(),
        _ => {
            let (y, m, d) = civil_from_days(t.div_euclid(86400));
            if y == civil_from_days(now.div_euclid(86400)).0 { format!("{} {d}", MON[m as usize - 1]) } else { y.to_string() }
        }
    }
}

// Howard Hinnant's civil-date algorithms.
fn days_from_civil(y: i64, m: i64, d: i64) -> i64 {
    let y = if m <= 2 { y - 1 } else { y };
    let era = y.div_euclid(400);
    let yoe = y - era * 400;
    let doy = (153 * (if m > 2 { m - 3 } else { m + 9 }) + 2) / 5 + d - 1;
    let doe = yoe * 365 + yoe / 4 - yoe / 100 + doy;
    era * 146_097 + doe - 719_468
}

fn civil_from_days(z: i64) -> (i64, i64, i64) {
    let z = z + 719_468;
    let era = z.div_euclid(146_097);
    let doe = z - era * 146_097;
    let yoe = (doe - doe / 1460 + doe / 36524 - doe / 146_096) / 365;
    let doy = doe - (365 * yoe + yoe / 4 - yoe / 100);
    let mp = (5 * doy + 2) / 153;
    let d = doy - (153 * mp + 2) / 5 + 1;
    let m = if mp < 10 { mp + 3 } else { mp - 9 };
    (yoe + era * 400 + i64::from(m <= 2), m, d)
}
```

```rust
//! Actions, their default keys, and keys.toml overrides.

use ratatui::crossterm::event::{KeyCode, KeyEvent, KeyModifiers};
use std::collections::HashMap;
use toml::Table;

#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
pub enum Action {
    Up, Down, Top, Bottom, Open, Back, NextPane, Search, Reply, ReplyAll, Forward, Compose,
    Star, ToggleRead, Delete, Move, Archive, Palette, Themes, Help, Quit,
}

/// (keys.toml name, default key, action, label)
pub const DEFAULTS: [(&str, &str, Action, &str); 21] = [
    ("up", "k", Action::Up, "move up"),
    ("down", "j", Action::Down, "move down"),
    ("top", "g", Action::Top, "top"),
    ("bottom", "G", Action::Bottom, "bottom"),
    ("open", "enter", Action::Open, "open"),
    ("back", "esc", Action::Back, "back"),
    ("next_pane", "tab", Action::NextPane, "next pane"),
    ("search", "/", Action::Search, "search"),
    ("reply", "r", Action::Reply, "reply"),
    ("reply_all", "R", Action::ReplyAll, "reply all"),
    ("forward", "f", Action::Forward, "forward"),
    ("compose", "c", Action::Compose, "compose"),
    ("star", "s", Action::Star, "star"),
    ("toggle_read", "u", Action::ToggleRead, "toggle read"),
    ("delete", "d", Action::Delete, "delete"),
    ("move", "m", Action::Move, "move to…"),
    ("archive", "a", Action::Archive, "archive"),
    ("palette", "ctrl+p", Action::Palette, "command palette"),
    ("themes", "t", Action::Themes, "themes"),
    ("help", "?", Action::Help, "help"),
    ("quit", "q", Action::Quit, "quit"),
];

pub struct Keymap {
    map: HashMap<(KeyCode, KeyModifiers), Action>,
    labels: HashMap<Action, String>,
}

impl Keymap {
    /// Defaults, then each `name = "key"` from keys.toml replaces that action's
    /// key (and wins any clash). Arrows, Home and End always move.
    pub fn new(overrides: &Table) -> Result<Self, String> {
        if let Some(bad) = overrides.keys().find(|k| !DEFAULTS.iter().any(|d| d.0 == k.as_str())) {
            return Err(format!("keys.toml: unknown action `{bad}`"));
        }
        let (mut map, mut labels) = (HashMap::new(), HashMap::new());
        for overridden in [false, true] {
            for (name, default, action, _) in DEFAULTS {
                let spec = match overrides.get(name) {
                    Some(toml::Value::String(s)) if overridden => s.as_str(),
                    Some(toml::Value::String(_)) => continue,
                    Some(_) => return Err(format!("keys.toml: `{name}` must be a string")),
                    None if overridden => continue,
                    None => default,
                };
                let key = parse(spec).ok_or_else(|| format!("keys.toml: can't parse key {spec:?} for `{name}`"))?;
                map.insert(key, action);
                labels.insert(action, spec.to_string());
            }
        }
        for (code, a) in [(KeyCode::Up, Action::Up), (KeyCode::Down, Action::Down), (KeyCode::Home, Action::Top), (KeyCode::End, Action::Bottom)] {
            map.entry((code, KeyModifiers::NONE)).or_insert(a);
        }
        Ok(Self { map, labels })
    }

    pub fn action(&self, k: KeyEvent) -> Option<Action> {
        // An uppercase char already implies shift; drop it so "G" matches.
        let mods = if matches!(k.code, KeyCode::Char(_)) { k.modifiers.difference(KeyModifiers::SHIFT) } else { k.modifiers };
        self.map.get(&(k.code, mods)).copied()
    }

    pub fn key_for(&self, a: Action) -> String {
        self.labels.get(&a).cloned().unwrap_or_default()
    }
}

pub fn parse(s: &str) -> Option<(KeyCode, KeyModifiers)> {
    let (mods, key) = if let Some(k) = s.strip_prefix("ctrl+") {
        (KeyModifiers::CONTROL, k)
    } else if let Some(k) = s.strip_prefix("alt+") {
        (KeyModifiers::ALT, k)
    } else {
        (KeyModifiers::NONE, s)
    };
    let code = match key {
        "enter" => KeyCode::Enter,
        "esc" => KeyCode::Esc,
        "tab" => KeyCode::Tab,
        "backtab" => KeyCode::BackTab,
        "space" => KeyCode::Char(' '),
        "up" => KeyCode::Up,
        "down" => KeyCode::Down,
        "left" => KeyCode::Left,
        "right" => KeyCode::Right,
        "home" => KeyCode::Home,
        "end" => KeyCode::End,
        "pgup" => KeyCode::PageUp,
        "pgdown" => KeyCode::PageDown,
        "backspace" => KeyCode::Backspace,
        "delete" => KeyCode::Delete,
        k if k.chars().count() == 1 => KeyCode::Char(k.chars().next()?),
        _ => return None,
    };
    Some((code, mods))
}
```

```rust
//! Hand a draft to $EDITOR. Drafts live under the state dir until a send succeeds.

use std::{fs, path::{Path, PathBuf}, process::Command, time::{SystemTime, UNIX_EPOCH}};

/// $VISUAL / $EDITOR may carry arguments ("code -w"), so it runs through sh
/// with the draft path passed safely as "$1".
pub fn edit(path: &Path) -> Result<(), String> {
    let editor = std::env::var("VISUAL").or_else(|_| std::env::var("EDITOR")).unwrap_or_else(|_| "vi".into());
    let st = Command::new("sh")
        .arg("-c")
        .arg(format!("{editor} \"$1\""))
        .arg("sh")
        .arg(path)
        .status()
        .map_err(|e| format!("{editor}: {e}"))?;
    if st.success() { Ok(()) } else { Err(format!("{editor} exited with {st}")) }
}

pub fn save(text: &str) -> Result<PathBuf, String> {
    let dir = crate::config::state_dir().join("drafts");
    fs::create_dir_all(&dir).map_err(|e| e.to_string())?;
    let ts = SystemTime::now().duration_since(UNIX_EPOCH).map(|d| d.as_nanos()).unwrap_or(0);
    let path = dir.join(format!("{ts}.eml"));
    fs::write(&path, text).map_err(|e| format!("{}: {e}", path.display()))?;
    Ok(path)
}
```

```rust
pub enum Done {
    Mailboxes(Result<Vec<String>, String>),

    Read { id: String, res: Result<String, String> },

impl Job {
    fn run(self, be: &Backend) -> Done {
        match self {
            Job::Mailboxes => Done::Mailboxes(be.mailboxes()),
            Job::List { mailbox, query, page } => {
                let res = match &query { Some(q) => be.search(&mailbox, q, page), None => be.list(&mailbox, page) };
                Done::List { mailbox, query, page, res }
            }
            Job::Read { mailbox, id } => {
                let res = be.read(&mailbox, &id);
                Done::Read { id, res }
            }
            Job::Triage { mailbox, id, op, undo } => {
                let res = match op {
                    Op::Flag { flag, on } => be.set_flag(&mailbox, &id, flag, on),
                    Op::Move { target } => be.move_to(&mailbox, &id, &target),
                    Op::Delete => be.delete(&mailbox, &id),
                };
                Done::Triage { mailbox, undo, res }
            }
            Job::Draft { kind, mailbox, id } => Done::Draft(be.draft(kind, &mailbox, &id)),
            Job::Send { path } => {
                let res = fs::read_to_string(&path).map_err(|e| e.to_string()).and_then(|raw| be.send(&raw));
                Done::Sent { path, res }
            }
        }
    }
}

    fn done(&mut self, d: Done) -> Option<Effect> {

                    self.mbox_sel = i;

                    return None; // a stale page: the user moved on

                self.loading = false;

                        self.page = page;

                    self.bodies.pop_front();
                }

                    self.reading = Some((id, body));
                    self.scroll = 0;
                }

            Done::Triage { res: Ok(()), .. } => {}
            Done::Triage { mailbox, undo, res: Err(e) } => {

                    self.revert(undo);
                }
                self.error(e);

            Done::Draft(Ok(text)) => match compose::save(&text) {
                Ok(path) => return Some(Effect::Edit(path)),
                Err(e) => self.error(e),
            },
            Done::Sent { path, res: Ok(()) } => {
```

```rust
pub mod app;
pub mod compose;
pub mod date;
pub mod keys;
pub mod view;
#[cfg(test)]
pub mod testutil;

use crate::config::Config;
use app::{App, Effect, Msg};
use ratatui::crossterm::{event::{self, DisableMouseCapture, EnableMouseCapture}, execute};
use std::{io, sync::{Arc, atomic::{AtomicBool, Ordering}, mpsc::{self, Sender}}, thread, time::Duration};

pub fn run(cfg: Config) -> Result<(), String> {
    let (tx, rx) = mpsc::channel();
    let mut app = App::new(cfg, tx.clone())?;
    let paused = Arc::new(AtomicBool::new(false));
    spawn_input(tx.clone(), paused.clone());
    spawn_ticker(tx.clone());
    let mut term = ratatui::init();
    let _ = execute!(io::stdout(), EnableMouseCapture);
    app.start();
    let res = (|| -> Result<(), String> {
        loop {
            term.draw(|f| view::draw(f, &mut app)).map_err(|e| e.to_string())?;
            let first = rx.recv().map_err(|e| e.to_string())?;
            for msg in std::iter::once(first).chain(rx.try_iter()) {
                let mut effect = app.update(msg);
                while let Some(Effect::Edit(path)) = effect {
                    // Hand the terminal to $EDITOR; the input thread pauses so it doesn't eat keys.
                    paused.store(true, Ordering::SeqCst);
                    let _ = execute!(io::stdout(), DisableMouseCapture);
                    ratatui::restore();
                    let r = compose::edit(&path);
                    term = ratatui::init();
                    let _ = execute!(io::stdout(), EnableMouseCapture);
                    paused.store(false, Ordering::SeqCst);
                    app.edited(path, r);
                    effect = None;
                }
            }
            if app.quit {
                return Ok(());
            }
        }
    })();
    let _ = execute!(io::stdout(), DisableMouseCapture);
    ratatui::restore();
    res
}

// ponytail: 50 ms poll; a key typed within 50 ms of the editor opening can be lost.
fn spawn_input(tx: Sender<Msg>, paused: Arc<AtomicBool>) {
    thread::spawn(move || loop {
        if paused.load(Ordering::SeqCst) {
            thread::sleep(Duration::from_millis(50));
            continue;
        }
        match event::poll(Duration::from_millis(50)) {
            Ok(true) => {
                if let Ok(ev) = event::read() {
                    if tx.send(Msg::Input(ev)).is_err() {
                        return;
                    }
                }
            }
            Ok(false) => {}
            Err(_) => return,
        }
    });
}

fn spawn_ticker(tx: Sender<Msg>) {
    thread::spawn(move || loop {
        thread::sleep(Duration::from_millis(100));
        if tx.send(Msg::Tick).is_err() {
            return;
        }
    });
}
```

```rust
/// Case-insensitive subsequence match; lower = tighter. None = no match.
pub fn fuzzy(query: &str, cand: &str) -> Option<usize> {
    let c: Vec<char> = cand.to_lowercase().chars().collect();
    let (mut pos, mut score) = (0, 0);
    for q in query.to_lowercase().chars().filter(|c| !c.is_whitespace()) {
        let i = c.get(pos..)?.iter().position(|&x| x == q)?;
        score += i;
        pos += i + 1;
    }
    Some(score)
}

/// Indexes of matching items, best first (stable for ties).
pub fn filter(items: &[String], q: &str) -> Vec<usize> {
    let mut v: Vec<(usize, usize)> = items.iter().enumerate().filter_map(|(i, s)| fuzzy(q, s).map(|sc| (sc, i))).collect();
    v.sort();
    v.into_iter().map(|(_, i)| i).collect()
}
```

```rust
    pub fn mouse(&mut self, m: MouseEvent) {
        let Some(&(pane, r)) = self.rects.iter().find(|(_, r)| r.contains(Position::new(m.column, m.row))) else { return };
        match m.kind {
            MouseEventKind::ScrollDown | MouseEventKind::ScrollUp => {
                self.focus = pane;
                self.step(if m.kind == MouseEventKind::ScrollDown { 3 } else { -3 });
            }
            MouseEventKind::Down(MouseButton::Left) => {
                self.focus = pane;
                let top = r.y + 1; // the title row
                if m.row < top {
                    return;
                }
                let row = (m.row - top) as usize;
                match pane {
                    // ponytail: mailbox lists are short, so no scroll offset is tracked there.
                    Pane::Mailboxes if row < self.mailboxes.len() => self.mbox_sel = row,
                    Pane::List if self.list_state.offset() + row < self.envelopes.len() => {
                        self.list_sel = self.list_state.offset() + row;
                        self.prefetch();
                    }
                    _ => return,
                }
                let now = Instant::now();
                let double = self.last_click.is_some_and(|(t, x, y)| now.duration_since(t).as_millis() < 400 && (x, y) == (m.column, m.row));
                self.last_click = if double { None } else { Some((now, m.column, m.row)) };
                if double {
                    self.open();
                }
            }
            _ => {}
        }
    }
```

```rust
/// `parcel omarchy hook install` makes Omarchy send SIGUSR1 on theme change.
#[cfg(unix)]
fn spawn_signals(tx: Sender<Msg>) {
    use signal_hook::{consts::SIGUSR1, iterator::Signals};
    if let Ok(mut signals) = Signals::new([SIGUSR1]) {
        thread::spawn(move || {
            for _ in signals.forever() {
                if tx.send(Msg::Repaint).is_err() {
                    return;
                }
            }
        });
    }
}
```

```sh
# Install parcel: curl -fsSL https://raw.githubusercontent.com/lubabs770/parcel/main/install.sh | sh
set -eu

case "$(uname -s)-$(uname -m)" in
  Darwin-arm64) target=aarch64-apple-darwin ;;
  Darwin-x86_64) target=x86_64-apple-darwin ;;
  Linux-x86_64) target=x86_64-unknown-linux-musl ;;
  Linux-aarch64 | Linux-arm64) target=aarch64-unknown-linux-musl ;;
  *) echo "parcel: no prebuilt binary for $(uname -s)-$(uname -m); try: cargo install --git https://github.com/lubabs770/parcel" >&2; exit 1 ;;
esac
if [ "$VERSION" = latest ]; then
  base="https://github.com/lubabs770/parcel/releases/latest/download"
else
  base="https://github.com/lubabs770/parcel/releases/download/$VERSION"
fi
asset="parcel-$target.tar.gz"
tmp=$(mktemp -d)
trap 'rm -rf "$tmp"' EXIT
curl -fsSL "$base/$asset" -o "$tmp/$asset"
curl -fsSL "$base/checksums.txt" -o "$tmp/checksums.txt"
# Verify before touching anything installed.
want=$(grep " $asset\$" "$tmp/checksums.txt" | cut -d' ' -f1)
if command -v sha256sum >/dev/null 2>&1; then got=$(sha256sum "$tmp/$asset" | cut -d' ' -f1)
else got=$(shasum -a 256 "$tmp/$asset" | cut -d' ' -f1); fi

"$tmp/parcel" --version >/dev/null || { echo "parcel: downloaded binary doesn't run here" >&2; exit 1; }
mkdir -p "$BIN_DIR"

echo "installed $("$BIN_DIR/parcel" --version) to $BIN_DIR/parcel"
```

```rust
//! `parcel upgrade`: re-run the one-line installer into wherever this binary lives.
//! The installer verifies checksums and test-runs the download before replacing anything.

use std::process::Command;

pub const INSTALL_URL: &str = "https://raw.githubusercontent.com/lubabs770/parcel/main/install.sh";
const LATEST_API: &str = "https://api.github.com/repos/lubabs770/parcel/releases/latest";

pub fn latest_tag() -> Result<String, String> {
    let out = Command::new("curl")
        .args(["-fsSL", "-H", "Accept: application/vnd.github+json", LATEST_API])
        .output()
        .map_err(|e| format!("curl: {e}"))?;
    if !out.status.success() {
        return Err(format!("couldn't reach GitHub: {}", String::from_utf8_lossy(&out.stderr).trim()));
    }
    let v: serde_json::Value = serde_json::from_slice(&out.stdout).map_err(|e| e.to_string())?;
    v["tag_name"].as_str().map(String::from).ok_or_else(|| "no release published yet".into())
}

pub fn newer(current: &str, tag: &str) -> bool {
    let parse = |v: &str| {
        let mut it = v.trim_start_matches('v').split(['.', '-']).map(|p| p.parse::<u64>().unwrap_or(0));
        (it.next().unwrap_or(0), it.next().unwrap_or(0), it.next().unwrap_or(0))
    };
    parse(tag) > parse(current)
}

pub fn run(check: bool) -> Result<(), String> {
    let current = env!("CARGO_PKG_VERSION");
    let tag = latest_tag()?;
    if !newer(current, &tag) {
        println!("parcel v{current} is up to date");
        return Ok(());
    }
    if check {
        println!("{tag} is available (you have v{current}). Run `parcel upgrade`.");
        return Ok(());
    }
    let exe = std::env::current_exe().map_err(|e| e.to_string())?;
    let dir = exe.parent().ok_or("can't find this binary's directory")?;
    // The URL is a constant passed as $1, never interpolated into the script.
    let st = Command::new("sh")
        .arg("-c")
        .arg("curl -fsSL \"$1\" | sh")
        .arg("sh")
        .arg(INSTALL_URL)
        .env("BIN_DIR", dir)
        .env("VERSION", &tag)
        .status()
        .map_err(|e| e.to_string())?;
    if st.success() { Ok(()) } else { Err(format!("installer exited with {st}")) }
}
```
